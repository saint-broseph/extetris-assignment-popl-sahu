engine

engine.py -- the headless ExTetris engine (v1 core, layer 3).

Pure Python: no tkinter, no timers, no threads, no file I/O.

  • Time enters only through tick(elapsed_ms).
  • Input enters only through apply_action(action).
  • Both return a list of Events, and the current GameState is an immutable snapshot.

Configuration is layered; later layers override earlier ones:

DEFAULTS  ->  spec (the game variant)  ->  player (preferences)  ->  runtime (live changes)
  • spec Any known key. Defines the variant; its SHA-256 is the variant hash.
  • player Only PLAYER_KEYS, e.g. UI settings, preview size, start level.
  • runtime Only RUNTIME_KEYS. Changes are requested with request_update() and applied at the next tick boundary, never in the middle of a move.

Triggers are restricted, data-only rules. No eval: conditions come from a fixed whitelist, and actions may only set RUNTIME_KEYS. Each trigger fires once:

{"id": "wrap-at-20", "when": {"lines_at_least": 20}, "set": {"rules.walls": "wrap"}}

Save and resume use JSON. The file stores the variant hash and a SHA-256 checksum of its own payload. Resuming against a different variant, or from an edited file, is refused.

 81def _canonical(obj: Any) -> str:
 82    return json.dumps(obj, sort_keys=True, separators=(",", ":"), default=list)

 85def _sha256(obj: Any) -> str:
 86    return hashlib.sha256(_canonical(obj).encode("utf-8")).hexdigest()

 89def _allowed(path: str, allowed: tuple[str, ...]) -> bool:
 90    return any(path == a or path.startswith(a + ".") for a in allowed)

 93def _set_path(d: dict, path: str, value: Any) -> None:
 94    parts = path.split(".")
 95    for p in parts[:-1]:
 96        d = d.setdefault(p, {})
 97    d[parts[-1]] = copy.deepcopy(value)

100def _merge(base: dict, over: Mapping) -> dict:
101    out = copy.deepcopy(base)
102    for k, v in over.items():
103        if k in ("pieces", "distribution", "triggers"):          # replaced as a whole
104            out[k] = copy.deepcopy(v)
105        elif isinstance(v, Mapping) and isinstance(out.get(k), dict):
106            out[k] = _merge(out[k], v)
107        else:
108            out[k] = copy.deepcopy(v)
109    return out

112def _check_known(layer: Mapping, name: str) -> None:
113    for section, value in layer.items():
114        if section not in DEFAULTS:
115            raise ConfigError(f"{name}: unknown section [{section}]; valid: {sorted(DEFAULTS)}")
116        if section in ("rules", "level", "variant") and isinstance(value, Mapping):
117            extra = set(value) - set(DEFAULTS[section])
118            if extra:
119                raise ConfigError(f"{name}: [{section}] unknown keys {sorted(extra)}")
120
121
122def _flatten_layer(layer: Mapping) -> list[str]:
123    paths = []
124    for section, value in layer.items():
125        if isinstance(value, Mapping) and section in ("rules", "level", "variant"):
126            paths += [f"{section}.{k}" for k in value]
127        else:
128            paths.append(section)
129    return paths

132def variant_hash(spec: Mapping | None) -> str:
133    """SHA-256 of the effective variant (defaults merged with spec)."""
134    return _sha256(_merge(DEFAULTS, spec or {}))

137def load_spec(path: str | Path) -> dict:
138    """Read a variant file (.toml via tomllib, or .json)."""
139    path = Path(path)
140    text = path.read_text(encoding="utf-8")
141    if path.suffix.lower() == ".toml":
142        import tomllib
143        return tomllib.loads(text)
144    return json.loads(text)

148    version, internal, gauss = rng.getstate()
149    return [version, list(internal), gauss]

159class Engine:
160    """Headless, deterministic game engine.
161
162    >>> e = Engine({"pieces": {"preset": "Full"}}, seed=7)
163    >>> events = e.apply_action(Action.HARD_DROP)
164    >>> events = e.tick(1000)
165    """
166
167    def __init__(self, spec: Mapping | None = None, *, player: Mapping | None = None,
168                 seed: int | None = None) -> None:
169        self._spec = copy.deepcopy(dict(spec or {}))
170        self._player = copy.deepcopy(dict(player or {}))
171        self._runtime: dict = {}
172        self._pending: list[tuple[str, Any]] = []
173        _check_known(self._spec, "spec")
174        _check_known(self._player, "player")
175        for p in _flatten_layer(self._player):
176            if not _allowed(p, PLAYER_KEYS):
177                raise ConfigError(f"player layer may not set '{p}'; allowed: {PLAYER_KEYS}")
178        self.variant_hash = variant_hash(self._spec)
179        self._build()
180        self._validate_triggers()
181        self.reset(seed)
182
183    # -- configuration ------------------------------------------------------------------
184    def _effective(self, runtime: Mapping | None = None) -> dict:
185        cfg = _merge(DEFAULTS, self._spec)
186        cfg = _merge(cfg, self._player)
187        return _merge(cfg, self._runtime if runtime is None else runtime)
188
189    def _components(self, cfg: Mapping):
190        try:
191            rules = Rules.from_dict(cfg["rules"])
192            scoring = Scoring.from_dict(cfg["scoring"])
193            level = LevelPolicy.from_dict(cfg["level"])
194            family = PieceFamily.from_spec(cfg["pieces"])
195            dist = distribution_from_spec(cfg["distribution"], family)
196            dist.probabilities()                     # fails early if every weight is zero
197        except PieceSpecError as exc:
198            raise ConfigError(str(exc)) from None
199        return rules, scoring, level, family, dist
200
201    def _build(self) -> None:
202        cfg = self._effective()
203        self.rules, self.scoring, self.level_policy, self.family, self.distribution = self._components(cfg)
204        self._config = cfg
205
206    @property
207    def config(self) -> dict:
208        """Effective configuration (a copy; edit layers, not this)."""
209        return copy.deepcopy(self._config)
210
211    def _validate_triggers(self) -> None:
212        seen = set()
213        for t in self._config["triggers"]:
214            if not isinstance(t, Mapping) or set(t) - {"id", "when", "set"} or "id" not in t:
215                raise ConfigError(f"trigger must be {{id, when, set}}: {t!r}")
216            if t["id"] in seen:
217                raise ConfigError(f"duplicate trigger id {t['id']!r}")
218            seen.add(t["id"])
219            for cond, val in t.get("when", {}).items():
220                if cond not in TRIGGER_CONDITIONS or not isinstance(val, int):
221                    raise ConfigError(f"trigger {t['id']!r}: condition must be one of {TRIGGER_CONDITIONS} with an int")
222            for path in t.get("set", {}):
223                if not _allowed(path, RUNTIME_KEYS):
224                    raise ConfigError(f"trigger {t['id']!r} may not set '{path}'; allowed: {RUNTIME_KEYS}")
225            trial = copy.deepcopy(self._runtime)
226            for path, value in t.get("set", {}).items():
227                _set_path(trial, path, value)
228            self._components(self._effective(trial))     # the trigger's result must be a valid config
229
230    def request_update(self, path: str, value: Any) -> None:
231        """Queue a runtime change; it is validated now and applied at the next tick boundary."""
232        if not _allowed(path, RUNTIME_KEYS):
233            raise ConfigError(f"'{path}' cannot change during a game; allowed: {RUNTIME_KEYS}")
234        trial = copy.deepcopy(self._runtime)
235        for p, v in self._pending:
236            _set_path(trial, p, v)
237        _set_path(trial, path, value)
238        self._components(self._effective(trial))
239        self._pending.append((path, copy.deepcopy(value)))
240
241    def _apply_updates(self, updates: list[tuple[str, Any]]) -> list[Event]:
242        if not updates:
243            return []
244        for path, value in updates:
245            _set_path(self._runtime, path, value)
246        self._build()
247        return [Event("ConfigChanged", {"paths": [p for p, _ in updates]})]
248
249    # -- lifecycle ----------------------------------------------------------------------------
250    def reset(self, seed: int | None = None) -> list[Event]:
251        """Start a new game (a new seed is derived deterministically if none is given)."""
252        if seed is None:
253            seed = self._rng.randrange(2 ** 63) if hasattr(self, "_rng") else secrets.randbits(63)
254        self.seed = int(seed)
255        self._rng = random.Random(self.seed)
256        start_level = self.level_policy.start_level
257        state = GameState(board=Board(self.rules.width, self.rules.height), active=None, pos=(0, 0),
258                          queue=(), mode=Mode.RUNNING, level=start_level)
259        for _ in range(self.rules.preview):
260            state = self._draw_into_queue(state)
261        self.state, events = self._spawn(state)
262        return [Event("Reset", {"seed": self.seed})] + events
263
264    def _draw_into_queue(self, s: GameState) -> GameState:
265        piece = self.distribution.draw(self._rng, s.history, s.draws)
266        return s.with_(queue=s.queue + (piece,), draws=s.draws + 1,
267                       history=(s.history + (piece.code,))[-HISTORY_CAP:])
268
269    def _spawn(self, s: GameState) -> tuple[GameState, list[Event]]:
270        while len(s.queue) < self.rules.preview + 1:
271            s = self._draw_into_queue(s)
272        piece, queue = s.queue[0], s.queue[1:]
273        pos = self.rules.spawn_position(piece)
274        s = s.with_(active=piece, pos=pos, queue=queue, pieces_spawned=s.pieces_spawned + 1, gravity_acc=0.0)
275        if not s.board.fits(piece, pos, self.rules.wrap):
276            return s.with_(mode=Mode.GAME_OVER), [Event("Spawned", {"piece": piece.code}),
277                                                  Event("GameOver", {"reason": "block_out", "score": s.score})]
278        return s, [Event("Spawned", {"piece": piece.code})]
279
280    # -- time -------------------------------------------------------------------------------------
281    def tick(self, elapsed_ms: float) -> list[Event]:
282        """Advance time. Tick boundary: pending updates and triggers are applied here."""
283        if elapsed_ms < 0:
284            raise ValueError("elapsed_ms must be >= 0")
285        pending, self._pending = self._pending, []
286        events = self._apply_updates(pending)
287        s = self.state
288        if s.mode is not Mode.RUNNING:
289            return events
290        acc, steps = s.gravity_acc + elapsed_ms, 0
291        s = s.with_(tick=s.tick + 1)
292        while s.mode is Mode.RUNNING and acc >= self.level_policy.gravity_ms(s.level) and steps < 10_000:
293            acc -= self.level_policy.gravity_ms(s.level)
294            steps += 1
295            s, ev = self._step_down(s, locks=True)
296            events += ev
297        self.state = s.with_(gravity_acc=acc if s.mode is Mode.RUNNING else 0.0)
298        events += self._run_triggers()
299        return events
300
301    def _run_triggers(self) -> list[Event]:
302        s, events, updates = self.state, [], []
303        values = {"lines_at_least": s.lines, "level_at_least": s.level, "score_at_least": s.score,
304                  "pieces_at_least": s.pieces_spawned, "tick_at_least": s.tick}
305        fired = set(s.fired_triggers)
306        for t in self._config["triggers"]:
307            if t["id"] in fired:
308                continue
309            if all(values[c] >= v for c, v in t.get("when", {}).items()):
310                fired.add(t["id"])
311                updates += list(t.get("set", {}).items())
312                events.append(Event("TriggerFired", {"id": t["id"]}))
313        if events:
314            self.state = s.with_(fired_triggers=frozenset(fired))
315            events += self._apply_updates(updates)
316        return events
317
318    # -- input ------------------------------------------------------------------------------------
319    def apply_action(self, action: Action | str) -> list[Event]:
320        action = Action(action) if not isinstance(action, Action) else action
321        s = self.state
322        if action is Action.RESET:
323            return self.reset()
324        if action is Action.PAUSE:
325            if s.mode is Mode.RUNNING:
326                self.state = s.with_(mode=Mode.PAUSED)
327                return [Event("Paused")]
328            if s.mode is Mode.PAUSED:
329                self.state = s.with_(mode=Mode.RUNNING)
330                return [Event("Resumed")]
331            return []
332        if s.mode is not Mode.RUNNING:
333            return []                                          # paused / game over: ignore input
334        wrap = self.rules.wrap
335        if action in (Action.MOVE_LEFT, Action.MOVE_RIGHT):
336            dc = -1 if action is Action.MOVE_LEFT else 1
337            col = s.pos[1] + dc
338            if wrap:
339                col %= self.rules.width
340            if s.board.fits(s.active, (s.pos[0], col), wrap):
341                self.state = s.with_(pos=(s.pos[0], col))
342                return [Event("Moved", {"pos": (s.pos[0], col)})]
343            return []
344        if action is Action.SOFT_DROP:
345            s2, events = self._step_down(s, locks=True)
346            if events and events[0].kind == "Moved":
347                s2 = s2.with_(score=s2.score + self.scoring.soft_drop, gravity_acc=0.0)
348            self.state = s2
349            return events
350        if action is Action.HARD_DROP:
351            d = s.board.drop_distance(s.active, s.pos, wrap)
352            s = s.with_(pos=(s.pos[0] + d, s.pos[1]), score=s.score + d * self.scoring.hard_drop)
353            events = [Event("Dropped", {"rows": d})]
354            if self.rules.hard_drop_locks:
355                s, more = self._lock(s)
356                events += more
357            self.state = s
358            return events
359        if action in (Action.ROTATE_CW, Action.ROTATE_CCW):
360            if self.rules.rotation == "none" or (self.rules.rotation == "cw_only" and action is Action.ROTATE_CCW):
361                return []
362            turned = s.active.turned(1 if action is Action.ROTATE_CW else -1)
363            for dc in self.rules.kick_offsets():
364                col = s.pos[1] + dc
365                if wrap:
366                    col %= self.rules.width
367                if s.board.fits(turned, (s.pos[0], col), wrap):
368                    self.state = s.with_(active=turned, pos=(s.pos[0], col))
369                    return [Event("Rotated", {"turn": turned.turn, "kick": dc})]
370            return []
371        raise ValueError(f"unhandled action {action}")
372
373    # -- mechanics ---------------------------------------------------------------------------------
374    def _step_down(self, s: GameState, locks: bool) -> tuple[GameState, list[Event]]:
375        below = (s.pos[0] + 1, s.pos[1])
376        if s.board.fits(s.active, below, self.rules.wrap):
377            return s.with_(pos=below), [Event("Moved", {"pos": below})]
378        return self._lock(s) if locks else (s, [])
379
380    def _lock(self, s: GameState) -> tuple[GameState, list[Event]]:
381        board = s.board.place(s.active, s.pos, self.rules.wrap)
382        events = [Event("Locked", {"piece": s.active.code, "pos": s.pos})]
383        board, lines, chains = board.clear_rows(self.rules.gravity)
384        score, total = s.score, s.lines + lines
385        if lines:
386            score += self.scoring.for_clear(lines, chains, s.level)
387            events.append(Event("LinesCleared", {"lines": lines, "chains": chains}))
388        level = max(s.level, self.level_policy.level_for(total))
389        if level != s.level:
390            events.append(Event("LevelChanged", {"level": level, "gravity_ms": self.level_policy.gravity_ms(level)}))
391        s = s.with_(board=board, score=score, lines=total, level=level, last_clear=lines,
392                    max_chain=max(s.max_chain, chains), active=None)
393        if self.rules.game_over == "top_rows" and any(r < self.rules.danger_rows for r, _ in board.filled):
394            return s.with_(mode=Mode.GAME_OVER), events + [Event("GameOver", {"reason": "top_rows", "score": score})]
395        s, more = self._spawn(s)
396        return s, events + more

399    def active_cells(self) -> list[tuple[int, int]]:
400        s = self.state
401        return [] if s.active is None else s.board.cells_of(s.active, s.pos, self.rules.wrap)

403    def ghost_cells(self) -> list[tuple[int, int]]:
404        s = self.state
405        if s.active is None or s.mode is Mode.GAME_OVER:
406            return []
407        d = s.board.drop_distance(s.active, s.pos, self.rules.wrap)
408        return s.board.cells_of(s.active, (s.pos[0] + d, s.pos[1]), self.rules.wrap)

410    # -- save / resume -------------------------------------------------------------------------
411    def save(self) -> str:
412        body = {"format": SAVE_FORMAT, "variant_hash": self.variant_hash,
413                "variant_name": self._config["variant"].get("name", ""), "seed": self.seed,
414                "player": self._player, "runtime": self._runtime,
415                "pending": [[p, v] for p, v in self._pending],
416                "state": self.state.to_json(), "rng": _rng_to_json(self._rng)}
417        body["payload_sha256"] = _sha256(body)
418        return json.dumps(body, indent=1, sort_keys=True)

420    @classmethod
421    def resume(cls, text: str, spec: Mapping | None, *, player: Mapping | None = None) -> "Engine":
422        """Rebuild an engine from save(). `spec` must be the variant the game was played with."""
423        try:
424            body = json.loads(text)
425            claimed = body.pop("payload_sha256")
426        except (ValueError, KeyError, AttributeError) as exc:
427            raise SaveError(f"not an ExTetris save file: {exc}") from None
428        if body.get("format") != SAVE_FORMAT:
429            raise SaveError(f"unsupported save format {body.get('format')!r}")
430        if _sha256(body) != claimed:
431            raise SaveError("save file checksum mismatch: the file was modified")
432        if variant_hash(spec) != body["variant_hash"]:
433            raise VariantMismatchError(
434                f"save belongs to variant {body.get('variant_name')!r} with a different hash; "
435                "load it with the same variant spec")
436        eng = cls(spec, player=player if player is not None else body["player"], seed=body["seed"])
437        eng._runtime = body["runtime"]
438        eng._build()
439        eng._pending = [(p, v) for p, v in body["pending"]]
440        eng.state = GameState.from_json(body["state"])
441        eng._rng = _rng_from_json(body["rng"])
442        return eng

212    def _validate_triggers(self) -> None:
213        seen = set()
214        for t in self._config["triggers"]:
215            if not isinstance(t, Mapping) or set(t) - {"id", "when", "set"} or "id" not in t:
216                raise ConfigError(f"trigger must be {{id, when, set}}: {t!r}")
217            if t["id"] in seen:
218                raise ConfigError(f"duplicate trigger id {t['id']!r}")
219            seen.add(t["id"])
220            for cond, val in t.get("when", {}).items():
221                if cond not in TRIGGER_CONDITIONS or not isinstance(val, int):
222                    raise ConfigError(f"trigger {t['id']!r}: condition must be one of {TRIGGER_CONDITIONS} with an int")
223            for path in t.get("set", {}):
224                if not _allowed(path, RUNTIME_KEYS):
225                    raise ConfigError(f"trigger {t['id']!r} may not set '{path}'; allowed: {RUNTIME_KEYS}")
226            trial = copy.deepcopy(self._runtime)
227            for path, value in t.get("set", {}).items():
228                _set_path(trial, path, value)
229            self._components(self._effective(trial))     # the trigger's result must be a valid config

232        """Queue a runtime change; it is validated now and applied at the next tick boundary."""
233        if not _allowed(path, RUNTIME_KEYS):
234            raise ConfigError(f"'{path}' cannot change during a game; allowed: {RUNTIME_KEYS}")

237            _set_path(trial, p, v)

282    def tick(self, elapsed_ms: float) -> list[Event]:
283        """Advance time. Tick boundary: pending updates and triggers are applied here."""
284        if elapsed_ms < 0:
285            raise ValueError("elapsed_ms must be >= 0")
286        pending, self._pending = self._pending, []
287        events = self._apply_updates(pending)
288        s = self.state
289        if s.mode is not Mode.RUNNING:
290            return events
291        acc, steps = s.gravity_acc + elapsed_ms, 0
292        s = s.with_(tick=s.tick + 1)
293        while s.mode is Mode.RUNNING and acc >= self.level_policy.gravity_ms(s.level) and steps < 10_000:
294            acc -= self.level_policy.gravity_ms(s.level)
295            steps += 1
296            s, ev = self._step_down(s, locks=True)
297            events += ev
298        self.state = s.with_(gravity_acc=acc if s.mode is Mode.RUNNING else 0.0)
299        events += self._run_triggers()
300        return events

302    def _run_triggers(self) -> list[Event]:
303        s, events, updates = self.state, [], []
304        values = {"lines_at_least": s.lines, "level_at_least": s.level, "score_at_least": s.score,
305                  "pieces_at_least": s.pieces_spawned, "tick_at_least": s.tick}
306        fired = set(s.fired_triggers)
307        for t in self._config["triggers"]:
308            if t["id"] in fired:
309                continue
310            if all(values[c] >= v for c, v in t.get("when", {}).items()):
311                fired.add(t["id"])
312                updates += list(t.get("set", {}).items())
313                events.append(Event("TriggerFired", {"id": t["id"]}))
314        if events:
315            self.state = s.with_(fired_triggers=frozenset(fired))
316            events += self._apply_updates(updates)
317        return events

320    def apply_action(self, action: Action | str) -> list[Event]:
321        action = Action(action) if not isinstance(action, Action) else action
322        s = self.state
323        if action is Action.RESET:
324            return self.reset()
325        if action is Action.PAUSE:
326            if s.mode is Mode.RUNNING:
327                self.state = s.with_(mode=Mode.PAUSED)
328                return [Event("Paused")]
329            if s.mode is Mode.PAUSED:
330                self.state = s.with_(mode=Mode.RUNNING)
331                return [Event("Resumed")]
332            return []
333        if s.mode is not Mode.RUNNING:
334            return []                                          # paused / game over: ignore input
335        wrap = self.rules.wrap
336        if action in (Action.MOVE_LEFT, Action.MOVE_RIGHT):
337            dc = -1 if action is Action.MOVE_LEFT else 1
338            col = s.pos[1] + dc
339            if wrap:
340                col %= self.rules.width
341            if s.board.fits(s.active, (s.pos[0], col), wrap):
342                self.state = s.with_(pos=(s.pos[0], col))
343                return [Event("Moved", {"pos": (s.pos[0], col)})]
344            return []
345        if action is Action.SOFT_DROP:
346            s2, events = self._step_down(s, locks=True)
347            if events and events[0].kind == "Moved":
348                s2 = s2.with_(score=s2.score + self.scoring.soft_drop, gravity_acc=0.0)
349            self.state = s2
350            return events
351        if action is Action.HARD_DROP:
352            d = s.board.drop_distance(s.active, s.pos, wrap)
353            s = s.with_(pos=(s.pos[0] + d, s.pos[1]), score=s.score + d * self.scoring.hard_drop)
354            events = [Event("Dropped", {"rows": d})]
355            if self.rules.hard_drop_locks:
356                s, more = self._lock(s)
357                events += more
358            self.state = s
359            return events
360        if action in (Action.ROTATE_CW, Action.ROTATE_CCW):
361            if self.rules.rotation == "none" or (self.rules.rotation == "cw_only" and action is Action.ROTATE_CCW):
362                return []
363            turned = s.active.turned(1 if action is Action.ROTATE_CW else -1)
364            for dc in self.rules.kick_offsets():
365                col = s.pos[1] + dc
366                if wrap:
367                    col %= self.rules.width
368                if s.board.fits(turned, (s.pos[0], col), wrap):
369                    self.state = s.with_(active=turned, pos=(s.pos[0], col))
370                    return [Event("Rotated", {"turn": turned.turn, "kick": dc})]
371            return []
372        raise ValueError(f"unhandled action {action}")

375    def _step_down(self, s: GameState, locks: bool) -> tuple[GameState, list[Event]]:
376        below = (s.pos[0] + 1, s.pos[1])
377        if s.board.fits(s.active, below, self.rules.wrap):
378            return s.with_(pos=below), [Event("Moved", {"pos": below})]
379        return self._lock(s) if locks else (s, [])

381    def _lock(self, s: GameState) -> tuple[GameState, list[Event]]:
382        board = s.board.place(s.active, s.pos, self.rules.wrap)
383        events = [Event("Locked", {"piece": s.active.code, "pos": s.pos})]
384        board, lines, chains = board.clear_rows(self.rules.gravity)
385        score, total = s.score, s.lines + lines
386        if lines:
387            score += self.scoring.for_clear(lines, chains, s.level)
388            events.append(Event("LinesCleared", {"lines": lines, "chains": chains}))
389        level = max(s.level, self.level_policy.level_for(total))
390        if level != s.level:
391            events.append(Event("LevelChanged", {"level": level, "gravity_ms": self.level_policy.gravity_ms(level)}))
392        s = s.with_(board=board, score=score, lines=total, level=level, last_clear=lines,
393                    max_chain=max(s.max_chain, chains), active=None)
394        if self.rules.game_over == "top_rows" and any(r < self.rules.danger_rows for r, _ in board.filled):
395            return s.with_(mode=Mode.GAME_OVER), events + [Event("GameOver", {"reason": "top_rows", "score": score})]
396        s, more = self._spawn(s)
397        return s, events + more
