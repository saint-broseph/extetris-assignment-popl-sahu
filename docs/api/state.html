 55def _from_dict(cls, data: Mapping, section: str):
 56    names = {f.name for f in fields(cls)}
 57    extra = set(data) - names
 58    if extra:
 59        raise ConfigError(f"[{section}] unknown keys {sorted(extra)}; valid: {sorted(names)}")
 60    try:
 61        return cls(**dict(data))
 62    except (TypeError, ValueError) as exc:
 63        raise ConfigError(f"[{section}] {exc}") from None

 81    def __post_init__(self):
 82        checks = [
 83            (4 <= self.width <= 64, "width must be 4..64"),
 84            (4 <= self.height <= 128, "height must be 4..128"),
 85            (self.walls in ("solid", "wrap"), "walls must be 'solid' or 'wrap'"),
 86            (self.rotation in ("both", "cw_only", "none"), "rotation must be 'both', 'cw_only' or 'none'"),
 87            (self.gravity in GRAVITY_MODES, f"gravity must be one of {GRAVITY_MODES}"),
 88            (self.game_over in ("block_out", "top_rows"), "game_over must be 'block_out' or 'top_rows'"),

 94            (isinstance(self.kicks, bool) and isinstance(self.hard_drop_locks, bool), "kicks/hard_drop_locks must be booleans"),

119class Scoring:
120    line_points: tuple[int, ...] = (0, 100, 300, 500, 800)
121    extra_line_points: int = 300          # per line beyond the table (cascades can clear more than 4)
122    chain_bonus: int = 50                 # per chain step after the first
123    soft_drop: int = 1                    # per cell
124    hard_drop: int = 2                    # per cell
125    level_multiplier: bool = True         # multiply clears by (level + 1)
126
127    def __post_init__(self):
128        object.__setattr__(self, "line_points", tuple(int(x) for x in self.line_points))
129        if len(self.line_points) < 2 or any(x < 0 for x in self.line_points):
130            raise ValueError("line_points needs >= 2 non-negative entries")
131
132    @classmethod
133    def from_dict(cls, data: Mapping) -> "Scoring":
134        return _from_dict(cls, data, "scoring")
135
136    def for_clear(self, lines: int, chains: int, level: int) -> int:
137        if lines <= 0:
138            return 0
139        table = self.line_points
140        base = table[lines] if lines < len(table) else table[-1] + (lines - len(table) + 1) * self.extra_line_points
141        base += self.chain_bonus * max(0, chains - 1)
142        return base * (level + 1 if self.level_multiplier else 1)

146class LevelPolicy:
147    base_ms: float = 1000.0
148    factor: float = 0.9
149    lines_per_level: int = 10
150    min_ms: int = 50
151    start_level: int = 0
152    max_level: int = 99
153
154    def __post_init__(self):
155        if not (self.base_ms > 0 and 0 < self.factor <= 1 and self.lines_per_level >= 1
156                and 1 <= self.min_ms <= self.base_ms and 0 <= self.start_level <= self.max_level):
157            raise ValueError("invalid level policy values")
158
159    @classmethod
160    def from_dict(cls, data: Mapping) -> "LevelPolicy":
161        return _from_dict(cls, data, "level")
162
163    def level_for(self, total_lines: int) -> int:
164        """Computed from the total, so a multi-line clear can never skip a threshold (v0 BUG-06)."""
165        return min(self.max_level, self.start_level + total_lines // self.lines_per_level)
166
167    def gravity_ms(self, level: int) -> int:
168        """1000 * 0.9 ** level by default, never below min_ms (v0 BUG-07)."""
169        return max(self.min_ms, round(self.base_ms * self.factor ** level))
